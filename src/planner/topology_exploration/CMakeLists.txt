cmake_minimum_required(VERSION 3.8)
project(topology_exploration)
set(CMAKE_CXX_STANDARD 17)
add_compile_options(-Wall -Wextra)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(message_filters REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(plan_env REQUIRED)
find_package(trajectory_manager REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
find_package(OpenCV REQUIRED)
find_package(OsqpEigen REQUIRED)
include_directories(include ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
add_library(topology_local_map src/rolling_local_map.cpp src/local_planner.cpp src/graph_planner.cpp)
ament_target_dependencies(topology_local_map plan_env rclcpp)
target_link_libraries(topology_local_map ${PCL_LIBRARIES})
add_library(topology_snapshot STATIC src/snapshot.cpp)
ament_target_dependencies(topology_snapshot plan_env)
add_executable(topology_explorer src/explorer_node.cpp src/visualization.cpp)
get_filename_component(APEXNAV_PROJECT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
target_compile_definitions(topology_explorer PRIVATE APEXNAV_PROJECT_ROOT="${APEXNAV_PROJECT_ROOT}")
ament_target_dependencies(topology_explorer rclcpp std_msgs std_srvs nav_msgs geometry_msgs sensor_msgs visualization_msgs message_filters cv_bridge plan_env trajectory_manager)
target_link_libraries(topology_explorer topology_local_map topology_snapshot ${OpenCV_LIBRARIES} OsqpEigen::OsqpEigen)
install(TARGETS topology_explorer RUNTIME DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  enable_testing()
  add_executable(local_map_test tests/local_map_test.cpp)
  target_link_libraries(local_map_test topology_local_map)
  ament_target_dependencies(local_map_test rclcpp plan_env)
  add_test(NAME rolling_local_map COMMAND local_map_test)
  add_executable(graph_planner_test tests/graph_planner_test.cpp src/graph_planner.cpp)
  ament_target_dependencies(graph_planner_test plan_env)
  add_test(NAME topology_astar COMMAND graph_planner_test)
  add_executable(local_planner_test tests/local_planner_test.cpp src/local_planner.cpp)
  ament_target_dependencies(local_planner_test plan_env)
  add_test(NAME local_astar_guard COMMAND local_planner_test)
  add_executable(frontier_generation_test tests/frontier_generation_test.cpp src/local_planner.cpp)
  ament_target_dependencies(frontier_generation_test plan_env)
  add_test(NAME frontier_generation COMMAND frontier_generation_test)
  add_executable(snapshot_test tests/snapshot_test.cpp)
  target_link_libraries(snapshot_test topology_snapshot)
  ament_target_dependencies(snapshot_test plan_env)
  add_test(NAME snapshot_lifecycle COMMAND snapshot_test "${CMAKE_CURRENT_BINARY_DIR}/snapshot_fixture")
  add_test(NAME snapshot_json COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tests/check_snapshot_json.py" "${CMAKE_CURRENT_BINARY_DIR}/snapshot_fixture")
  set_tests_properties(snapshot_json PROPERTIES DEPENDS snapshot_lifecycle)
  add_executable(mpc_safety_test tests/mpc_safety_test.cpp)
  ament_target_dependencies(mpc_safety_test trajectory_manager rclcpp)
  target_link_libraries(mpc_safety_test OsqpEigen::OsqpEigen)
  add_test(NAME mpc_safety COMMAND mpc_safety_test)
  foreach(test local_map_test graph_planner_test local_planner_test mpc_safety_test frontier_generation_test snapshot_test)
    target_compile_options(${test} PRIVATE -UNDEBUG)
  endforeach()
  set_tests_properties(mpc_safety PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=94")
  set_tests_properties(rolling_local_map PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=94")
endif()
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
ament_package()
